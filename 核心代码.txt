5.1 用户登录模块核心代码

// LoginController.java - 登录验证与角色分发
@PostMapping("/doLogin")
public String doLogin(@RequestParam String username,
                      @RequestParam String password,
                      HttpSession session, Model model) {
    User user = userService.login(username, password);
    if (user == null) {
        model.addAttribute("error", "用户名或密码错误");
        return "login";
    }
    session.setAttribute("loginUser", user);
    switch (user.getRole()) {
        case 1: return "redirect:/admin/index";
        case 2: return "redirect:/teacher/index";
        case 3: return "redirect:/student/index";
    }
}

// UserServiceImpl.java - 密码MD5加密校验
@Override
public User login(String username, String password) {
    User user = userMapper.findByUsername(username);
    if (user == null || (user.getStatus() != null && user.getStatus() == 0))
        return null;
    String encrypted = MD5Util.encrypt(password);
    return encrypted.equals(user.getPassword()) ? user : null;
}


5.2 登录拦截器核心代码

// LoginInterceptor.java - Session校验与AJAX适配
@Override
public boolean preHandle(HttpServletRequest request,
        HttpServletResponse response, Object handler) throws Exception {
    HttpSession session = request.getSession();
    if (session != null && session.getAttribute("loginUser") != null)
        return true;
    boolean isAjax = "XMLHttpRequest"
            .equalsIgnoreCase(request.getHeader("X-Requested-With"));
    if (isAjax) {
        response.setContentType("application/json;charset=UTF-8");
        response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
        Result result = Result.error("未登录或登录已过期");
        response.getWriter().write(new ObjectMapper().writeValueAsString(result));
        return false;
    }
    response.sendRedirect(request.getContextPath() + "/login");
    return false;
}

// InternshipMapper.xml - 多表关联查询
<select id="findByCondition" resultMap="BaseResultMap">
    SELECT i.*, s.name AS student_name,
           c.name AS company_name, t.name AS teacher_name
    FROM internship i
    LEFT JOIN student s ON i.student_id = s.id
    LEFT JOIN company c ON i.company_id = c.id
    LEFT JOIN teacher t ON i.teacher_id = t.id
    <where>
        <if test="studentName != null and studentName != ''">
            AND s.name LIKE CONCAT('%', #{studentName}, '%')
        </if>
        <if test="status != null">AND i.status = #{status}</if>
    </where>
</select>


5.7 任务管理模块核心代码

// TeacherController.java - 教师布置/编辑任务
@PostMapping("/task/save")
public String taskSave(@ModelAttribute InternshipTask task, HttpSession session) {
    Teacher teacher = (Teacher) session.getAttribute("loginTeacher");
    Internship internship = internshipService.findById(task.getInternshipId());
    if (internship == null || !teacher.getId().equals(internship.getTeacherId()))
        return "redirect:/teacher/tasks";
    if (task.getId() == null) {
        task.setStatus(0);
        task.setCreateTime(new Date());
        taskService.addTask(task);
    } else {
        task.setUpdateTime(new Date());
        taskService.updateTask(task);
    }
    return "redirect:/teacher/tasks";
}

// StudentController.java - 学生更新任务状态
@PostMapping("/task/updateStatus")

public Result taskUpdateStatus(@RequestParam Integer taskId,
        @RequestParam Integer status, HttpSession session) {
    Student student = (Student) session.getAttribute("loginStudent");
    InternshipTask task = taskService.findById(taskId);
    task.setStatus(status);
    task.setUpdateTime(new Date());
    return taskService.updateTask(task) ? Result.success() : Result.error("更新失败");
}

// StudentController.java - 报告提交（含文件上传）
@PostMapping("/report/save")
public String reportSave(@ModelAttribute InternshipReport report,
        @RequestParam MultipartFile attachment, HttpSession session,
        HttpServletRequest request) throws IOException {
    Student student = (Student) session.getAttribute("loginStudent");
    report.setStudentId(student.getId());
    if (attachment != null && !attachment.isEmpty()) {
        String uploadPath = request.getServletContext().getRealPath("/upload/");
        String fileName = "report_" + System.currentTimeMillis() + suffix;
        attachment.transferTo(new File(uploadPath, fileName));
        report.setFilePath("/upload/" + fileName);
    }
    report.setStatus(0);
    report.setSubmitTime(new Date());
    reportService.addReport(report);
    return "redirect:/student/reports";
}

// TeacherController.java - 教师审阅报告
@PostMapping("/report/review")
public String reportReview(@RequestParam Integer id,
        @RequestParam Integer status, @RequestParam String feedback,
        @RequestParam Integer score, HttpSession session) {
    Teacher teacher = (Teacher) session.getAttribute("loginTeacher");
    InternshipReport report = reportService.findById(id);
    Internship internship = internshipService.findById(report.getInternshipId());
    if (teacher.getId().equals(internship.getTeacherId())) {
        reportService.reviewReport(id, status, feedback, score);
    }
    return "redirect:/teacher/reports";
}

5.11 密码修改模块核心代码

// PasswordController.java - 密码验证与更新
@PostMapping("/password/update")
@ResponseBody
public Result updatePassword(@RequestParam String oldPassword,
        @RequestParam String newPassword, HttpSession session) {
    User loginUser = (User) session.getAttribute("loginUser");
    if (loginUser == null) return Result.error("请先登录");
    boolean success = userService.updatePassword(
            loginUser.getId(), oldPassword, newPassword);
    return success ? Result.success("密码修改成功") : Result.error("原密码错误");
}

// UserServiceImpl.java - MD5加密比对
@Override
public boolean updatePassword(Integer id, String oldPwd, String newPwd) {
    User user = userMapper.findById(id);
    if (!MD5Util.encrypt(oldPwd).equals(user.getPassword())) return false;
    return userMapper.updatePassword(id, MD5Util.encrypt(newPwd)) > 0;
}

// AdminController.java - 统计数据接口
@GetMapping("/statistics/data")

    Map<String, Object> data = new HashMap<>();
